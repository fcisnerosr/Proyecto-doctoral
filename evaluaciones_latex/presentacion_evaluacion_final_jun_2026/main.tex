\begin{frame}{Formulación Matemática: Indicadores Modales ($DI_1 - DI_3$)}
    \small
    \textbf{1. COMAC ($DI_1$):}
    Evalúa la correlación punto a punto de los modos. (1 = Correlación perfecta, 0 = Sin correlación).
    \begin{equation}
        \text{DI}_1(j) = 1 - \sqrt{ \frac{ \left( \sum_{i=1}^{N_m} |\phi_{u,ij}| |\phi_{d,ij}| \right)^2 }{ \sum_{i=1}^{N_m} \phi_{u,ij}^2 \sum_{i=1}^{N_m} \phi_{d,ij}^2 } }
    \end{equation}

    \vspace{0.2cm}

    \textbf{2. Diferencia Absoluta ($DI_2$) y Razón Relativa ($DI_3$):}
    Cuantifican el desplazamiento de los vectores modales $\boldsymbol{\phi}$.
    \begin{align}
        \text{DI}_2(j) &= \mathcal{R}_j \left( \sum_{i=1}^{N_m} | \boldsymbol{\phi}_{u,i} - \boldsymbol{\phi}_{d,i} | \right) \\
        \text{DI}_3(j) &= \mathcal{R}_j \left( \sum_{i=1}^{N_m} \left| \frac{\boldsymbol{\phi}_{d,i}}{\boldsymbol{\phi}_{u,i} + \epsilon} - 1 \right| \right)
    \end{align}

    \footnotesize \textit{*Donde $u$ es estado intacto, $d$ dañado y $\epsilon$ es un factor de regularización.}
\end{frame}

\begin{frame}{Formulación Matemática: Flexibilidad y Estadística ($DI_4 - DI_8$)}
    \small
    \textbf{Basados en la Matriz de Flexibilidad ($\mathbf{F} \approx \boldsymbol{\Phi} \boldsymbol{\Omega}^{-2} \boldsymbol{\Phi}^T$):}
    \begin{itemize}
        \item \textbf{Diferencia ($DI_4$):} $| \text{diag}(\mathbf{F}_d) - \text{diag}(\mathbf{F}_u) |$
        \item \textbf{Razón ($DI_5$):} $\left| \frac{\text{diag}(\mathbf{F}_d)}{\text{diag}(\mathbf{F}_u) + \epsilon} - 1 \right|$
        \item \textbf{Variación \% ($DI_6$):} Versión porcentual de la razón ($DI_5 \times 100$).
    \end{itemize}

    \vspace{0.2cm}
    \hrule
    \vspace{0.2cm}

    \textbf{Estandarización Estadística ($DI_7$ y $DI_8$):}
    Normalizan el daño asumiendo una distribución Gaussiana para resaltar anomalías (outliers).
    \begin{equation}
        \text{DI}_7(j) = \frac{\Delta F_j - \mu_{\Delta F}}{\sigma_{\Delta F}} \quad \rightarrow \quad \text{DI}_8(j) = 1 - 2\left( 1 - \Phi_{\text{cdf}}(|\text{DI}_7|) \right)
    \end{equation}
\end{frame}

% ============= P3: PRUEBA DE WILCOXON =============

\begin{frame}{P1: Prueba de Wilcoxon — ICD vs Baseline (Descripción)}
    \small
    \begin{block}{Prueba de Significancia Estadística}
        Se realiza la prueba de rangos con signo de Wilcoxon para evaluar si el ICD fusionado es \textbf{estadísticamente superior} al mejor indicador individual ($DI_6$, baseline):
        \begin{itemize}
            \item $H_0$ (hipótesis nula): Mediana del ICD = Mediana del $DI_6$
            \item $H_1$ (hipótesis alternativa): Mediana del ICD $>$ Mediana del $DI_6$
        \end{itemize}
    \end{block}

    \vspace{0.3cm}

    \begin{block}{Resultado Clave}
        \textbf{$p$-value $< 0.01$} — confirmamos con significancia estadística del 99\% que el ICD fusionado supera significativamente al mejor indicador individual sobre todos los casos de daño.
    \end{block}
\end{frame}

\begin{frame}{P1: Fórmula — Prueba de Wilcoxon}
    \footnotesize
    \begin{block}{Estadístico de Prueba}
        El estadístico de Wilcoxon se calcula como:
        \begin{equation*}
            W = \sum_{i=1}^{n} \text{sgn}(X_i - Y_i) \cdot R_i
        \end{equation*}
        \vspace{0.25cm}
        \textbf{Definiciones:}
        \begin{itemize} \footnotesize
            \item $X_i$ = ICD para el caso $i$; $Y_i$ = $DI_6$ baseline
            \item $\text{sgn}(x)$ = función signo: devuelve $+1$ (ICD mejor), $-1$ ($DI_6$ mejor), $0$ (iguales)
            \item $R_i$ = posición (rango) de $|X_i - Y_i|$ al ordenar todas las diferencias de menor a mayor
        \end{itemize}
    \end{block}

    \vspace{0.15cm}

    \begin{block}{Resultados Detallados}

    \end{block}
\end{frame}

\begin{frame}{P1: Curva POD — ¿Qué es y cómo se calcula?}
    \small
    \begin{block}{Probability of Detection (POD)}
        Para cada nivel de severidad $a_i$ (\% de daño), la POD empírica se calcula como:
        \begin{equation*}
            \text{POD}(a_i) = \frac{\text{\# detecciones correctas en nivel } a_i}{\text{\# total de casos en nivel } a_i}
        \end{equation*}
        Las bandas de confianza al 95\% se calculan con el \textbf{intervalo de Wilson}:
        \begin{equation*}
            \hat{p} = \frac{p + \frac{z^2}{2n}}{1 + \frac{z^2}{n}} \pm \frac{z\sqrt{\frac{p(1-p)}{n} + \frac{z^2}{4n^2}}}{1 + \frac{z^2}{n}}, \quad z = 1.96
        \end{equation*}
    \end{block}
    \vspace{0.15cm}
    \footnotesize
    \textbf{Notas:} El \textbf{intervalo de Wilson (1927)} es el estándar recomendado (ASTM E2546) para CIs de proporciones en SHM/NDT, robusto con muestras pequeñas o proporciones cercanas a 0 o 1. \quad $z=1.96$ es el valor crítico de la normal estándar para 95\% de confianza (valor fijo y universal).
\end{frame}

\begin{frame}{P1: AUC-POD — ¿Qué es y cómo se calcula?}
    \small
    \begin{block}{AUC-POD: Área Bajo la Curva POD}
        Resume el desempeño global de detección en \textbf{un solo número}. Se calcula con la regla del trapecio, normalizada sobre el rango de severidades:
        \begin{equation*}
            \text{AUC-POD} = \frac{\int_{a_{\min}}^{a_{\max}} \text{POD}(a)\, da}{a_{\max} - a_{\min}}
        \end{equation*}
        \textbf{Rango:} 0 a 1. \quad \textbf{Interpretación:} AUC-POD $\approx 1$ = detección casi perfecta en todos los niveles de severidad.
    \end{block}
\end{frame}

\begin{frame}{P3: Precisión Topológica — Contribución Adicional a la Investigación}
    \footnotesize
    \begin{block}{Limitación de la Precisión Clásica}
        Penaliza todos los Falsos Positivos (FPs) por igual, sin importar su proximidad al elemento dañado:
        \begin{equation*}
            \text{Precision} = \frac{TP}{TP + FP}
        \end{equation*}
        En estructuras tipo \textit{lattice} (jacket), un FP adyacente ($d=1$) tiene interpretación muy diferente a uno al extremo opuesto ($d=9$).
    \end{block}

    \vspace{0.1cm}

    \begin{block}{Precisión Ponderada por Topología — Propuesta Novedosa}
        Asigna un peso $w(d_i) \in [0,1]$ a cada FP según su distancia topológica $d$ al elemento dañado (BFS sobre el grafo de conectividad):
        \begin{equation*}
            \text{Precision}_w = \frac{TP}{TP + \displaystyle\sum_{i \in FP} w(d_i)}
        \end{equation*}
    \end{block}
\end{frame}
