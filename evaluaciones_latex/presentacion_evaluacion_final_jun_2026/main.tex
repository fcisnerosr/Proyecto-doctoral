% ============= P3: PRUEBA DE WILCOXON =============

\begin{frame}{P3: Prueba de Wilcoxon — ICD vs Baseline (Descripción)}
    \small
    \begin{block}{Análisis No-Paramétrico de Significancia}
        Se realiza una prueba de rangos de Wilcoxon (test no-paramétrico) para evaluar si el ICD fusionado es \textbf{estadísticamente superior} al mejor indicador individual ($DI_6$, baseline):
        \begin{itemize}
            \item $H_0$ (hipótesis nula): Mediana del ICD = Mediana del $DI_6$
            \item $H_1$ (hipótesis alternativa): Mediana del ICD $>$ Mediana del $DI_6$
        \end{itemize}
    \end{block}

    \vspace{0.3cm}

    \begin{block}{Resultado Clave}
        \textbf{$p$-value $< 0.01$} — confirmamos con significancia estadística del 99\% que el ICD fusionado supera significativamente al mejor indicador individual sobre todos los casos de daño.
    \end{block}
\end{frame}

\begin{frame}{P3: Fórmula — Prueba de Wilcoxon}
    \footnotesize
    \begin{block}{Estadístico de Prueba}
        El estadístico de Wilcoxon se calcula como:
        \begin{equation*}
            W = \sum_{i=1}^{n} \text{sgn}(X_i - Y_i) \cdot R_i
        \end{equation*}
        \vspace{-0.3cm}
        donde: $X_i$ = ICD, $Y_i$ = $DI_6$ baseline, $R_i$ = rango de diferencias, $\text{sgn}(\cdot)$ = función signo
    \end{block}

    \vspace{0.15cm}

    \begin{block}{P-value}
        Si $p < 0.01$, se rechaza $H_0$ con confianza del 99\%.
    \end{block}
\end{frame}

% ============= P4: ROBUSTEZ AL RUIDO AWGN =============

\begin{frame}{P4: Robustez al Ruido AWGN (Descripción)}
    \small
    \begin{block}{Simulación de Condiciones Experimentales Realistas}
        Se evalúa la estabilidad del ICD bajo \textbf{ruido blanco Gaussiano aditivo (AWGN)}, simulando las imperfecciones inherentes de sensores y equipos de medición en campo:
        \begin{itemize}
            \item Se agrrega AWGN a las respuestas vibratorias sintéticas
            \item Se varía la relación señal-ruido (SNR) en tres niveles: \textbf{20, 30, 40 dB}
            \item Se mide la degradación de detectabilidad del ICD en cada nivel
        \end{itemize}
    \end{block}

    \vspace{0.3cm}

    \begin{block}{Hallazgo Crítico}
        El ICD \textbf{mantiene detectabilidad aceptable} incluso bajo SNR = 20 dB (condición severamente ruidosa), demostrando robustez práctica para aplicaciones de campo.
    \end{block}
\end{frame}

\begin{frame}{P4: Fórmula — SNR y Degradación}
    \small
    \begin{block}{Relación Señal-Ruido (SNR)}
        Se define formalmente como:
        \begin{equation*}
            \text{SNR}_{dB} = 10 \log_{10} \left( \frac{P_{\text{señal}}}{P_{\text{ruido}}} \right)
        \end{equation*}
        donde $P_{\text{señal}}$ y $P_{\text{ruido}}$ son potencias (varianzas en el caso de señales de media cero).
    \end{block}

    \vspace{0.2cm}

    \begin{block}{Respuesta Ruidosa}
        La respuesta vibradora contaminada es:
        \begin{equation*}
            \mathbf{y}_{\text{noisy}} = \mathbf{y}_{\text{limpia}} + \mathbf{n} \quad \text{donde} \quad \mathbf{n} \sim \mathcal{N}(0, \sigma_n^2)
        \end{equation*}
        y $\sigma_n^2$ se ajusta para lograr los SNR objetivo (20, 30, 40 dB).
    \end{block}
\end{frame}

% ============= P5: HUNGARIAN ALGORITHM & MÉTRICAS =============

\begin{frame}{P5: Hungarian Algorithm y Métricas de Clasificación (Descripción)}
    \small
    \begin{block}{Asignación Óptima de Detecciones}
        Se implementa el Algoritmo Húngaro (Hungarian Algorithm) para resolver el problema de asignación óptima:
        \begin{itemize}
            \item Empareja de forma biunívoca cada \textbf{detección predicha} con un \textbf{daño verdadero} (ground-truth)
            \item Minimiza la distancia topológica total
            \item Evita doble-conteo y garantiza un mapeo uno-a-uno
        \end{itemize}
    \end{block}

    \vspace{0.3cm}

    \begin{block}{Métricas Resultantes}
        Con la asignación óptima se calculan:
        \begin{itemize}
            \item \textbf{TP (Verdaderos Positivos):} detecciones correctas
            \item \textbf{FP (Falsos Positivos):} predicciones sin correspondencia
            \item \textbf{Precision, Recall, F1:} indicadores de desempeño global
        \end{itemize}
    \end{block}
\end{frame}

\begin{frame}{P5: Fórmula — Hungarian Algorithm}
    \small
    \begin{block}{Problema de Asignación}
        Sea $\mathcal{P} = \{p_1, \ldots, p_m\}$ el conjunto de predicciones y $\mathcal{G} = \{g_1, \ldots, g_n\}$ el ground-truth. Se busca:
        \begin{equation*}
            \sigma^* = \arg\min_{\sigma \in S_m} \sum_{i=1}^{\min(m,n)} d(p_i, g_{\sigma(i)})
        \end{equation*}
        donde $d(p_i, g_j)$ es la distancia Euclidiana entre nodos en el espacio 3D.
    \end{block}

    \vspace{0.2cm}

    \begin{block}{Métricas de Clasificación}
        \begin{equation*}
            \text{Precision} = \frac{TP}{TP + FP}, \quad \text{Recall} = \frac{TP}{TP + FN}, \quad F1 = 2 \cdot \frac{\text{Precision} \cdot \text{Recall}}{\text{Precision} + \text{Recall}}
        \end{equation*}
    \end{block}
\end{frame}

\begin{frame}{Procesador de Fatiga SACS v1.0}
    \begin{columns}[c]
        \column{0.5\textwidth}
        \centering
        \small
        \textbf{Contexto del Proyecto:}

        El Instituto Mexicano del Petróleo requería una herramienta para consolidar reportes de análisis de fatiga generados por SACS (Structural Analysis Computer System).

        \vspace{0.3cm}

        \textbf{Problema:} SACS NO suma automáticamente el daño acumulado entre diferentes periodos temporales — los ingenieros deben hacerlo manualmente, proceso tedioso y propenso a errores con miles de elementos.
